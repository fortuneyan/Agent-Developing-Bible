## 1.6 成本预估与管理

### 1.6.1 不同模型的成本对比

**主流模型定价(2026年)**:

| 模型 | 输入价格(/1K tokens) | 输出价格(/1K tokens) | 上下文长度 | 适用场景 |
|:---|:---:|:---:|:---:|:---|
| GPT-4o | $0.0025 | $0.01 | 128K | 复杂推理任务 |
| GPT-4o-mini | $0.00015 | $0.0006 | 128K | 日常对话、简单任务 |
| Claude 3.5 Sonnet | $0.003 | $0.015 | 200K | 长文档分析 |
| 文心一言 ERNIE-4.0 | ¥0.03 | ¥0.06 | 8K | 中文场景 |
| 通义千问 Max | ¥0.02 | ¥0.06 | 32K | 国产化需求 |

**成本计算公式**:
```
单次调用成本 = (输入tokens × 输入单价 + 输出tokens × 输出单价) / 1000

月度预估成本 = 日均调用量 × 30 × 单次平均成本
```

### 1.6.2 成本预估工具

```python
class CostEstimator:
    """LLM调用成本预估器"""
    
    PRICING = {
        "gpt-4o": {"input": 0.0025, "output": 0.01},
        "gpt-4o-mini": {"input": 0.00015, "output": 0.0006},
        "claude-3.5-sonnet": {"input": 0.003, "output": 0.015}
    }
    
    def __init__(self, model: str = "gpt-4o-mini"):
        self.model = model
        self.pricing = self.PRICING.get(model, {})
    
    def estimate_tokens(self, text: str) -> int:
        """估算文本的token数(简化版)"""
        # 中文约1.5字符/token,英文约4字符/token
        chinese_chars = sum(1 for c in text if '\u4e00' <= c <= '\u9fff')
        english_chars = len(text) - chinese_chars
        
        return int(chinese_chars / 1.5 + english_chars / 4)
    
    def calculate_cost(
        self, 
        input_text: str, 
        output_text: str
    ) -> dict:
        """计算单次调用成本"""
        input_tokens = self.estimate_tokens(input_text)
        output_tokens = self.estimate_tokens(output_text)
        
        cost = (
            input_tokens * self.pricing["input"] + 
            output_tokens * self.pricing["output"]
        ) / 1000
        
        return {
            "model": self.model,
            "input_tokens": input_tokens,
            "output_tokens": output_tokens,
            "total_tokens": input_tokens + output_tokens,
            "cost_usd": cost,
            "cost_cny": cost * 7.2  # 简化汇率
        }
    
    def estimate_monthly_cost(
        self,
        daily_queries: int,
        avg_input_length: int,
        avg_output_length: int
    ) -> dict:
        """预估月度成本"""
        single_cost = self.calculate_cost(
            "x" * avg_input_length,
            "y" * avg_output_length
        )
        
        monthly_cost = single_cost["cost_cny"] * daily_queries * 30
        
        return {
            "daily_queries": daily_queries,
            "single_query_cost": single_cost["cost_cny"],
            "monthly_cost": monthly_cost,
            "monthly_budget_needed": monthly_cost * 1.2  # 留20%buffer
        }


# 使用示例
estimator = CostEstimator("gpt-4o-mini")
print(estimator.estimate_monthly_cost(
    daily_queries=1000,
    avg_input_length=500,
    avg_output_length=300
))
# 输出: 月度成本约¥81, 建议预算¥97.2
```

class TieredModelSelector:
    """分级模型选择器: 根据任务复杂度选择合适的模型"""
    
    def select_model(self, query: str, complexity: str = "auto") -> str:
        """
        根据任务复杂度选择模型
        
        Args:
            query: 用户查询
            complexity: "simple" | "medium" | "complex" | "auto"
        """
        if complexity == "auto":
            complexity = self._analyze_complexity(query)
        
        if complexity == "simple":
            return "gpt-4o-mini"  # 简单任务用便宜模型
        elif complexity == "medium":
            return "gpt-4o-mini"  # 中等任务也用mini
        else:
            return "gpt-4o"  # 复杂任务才用强模型
    
    def _analyze_complexity(self, query: str) -> str:
        """分析任务复杂度"""
        # 简单启发式规则
        if len(query) < 50:
            return "simple"
        elif any(kw in query for kw in ["分析", "推理", "规划", "复杂"]):
            return "complex"
        else:
            return "medium"
```

class FailoverModelChain:
    """故障降级模型链"""
    
    def __init__(self):
        self.models = [
            {"name": "gpt-4o", "priority": 1, "status": "active"},
            {"name": "claude-3.5-sonnet", "priority": 2, "status": "active"},
            {"name": "qwen-max", "priority": 3, "status": "active"}  # 国产备用
        ]
    
    def generate(self, prompt: str) -> str:
        """按优先级尝试,失败则降级"""
        for model in self.models:
            if model["status"] != "active":
                continue
            
            try:
                return self._call_model(model["name"], prompt)
            except Exception as e:
                print(f"模型{model['name']}调用失败: {e}")
                model["status"] = "degraded"
                continue
        
        raise RuntimeError("所有模型均不可用")

## 4.6 工具安全性设计

### 4.6.1 安全威胁分析

**常见安全风险**:

1. **权限越权**: Agent调用用户无权访问的工具
2. **参数注入**: 恶意用户通过输入注入危险参数
3. **资源滥用**: Agent无限制调用昂贵工具
4. **数据泄露**: 工具返回敏感信息

### 4.6.2 权限控制实现

```python
from enum import Enum
from typing import Set

class UserRole(Enum):
    """用户角色"""
    GUEST = "guest"
    USER = "user"
    ADMIN = "admin"


class ToolPermissionManager:
    """工具权限管理器"""
    
    # 工具权限配置
    TOOL_PERMISSIONS = {
        "search_web": {"roles": {UserRole.GUEST, UserRole.USER, UserRole.ADMIN}},
        "delete_file": {"roles": {UserRole.ADMIN}},
        "send_email": {"roles": {UserRole.USER, UserRole.ADMIN}},
        "query_database": {"roles": {UserRole.USER, UserRole.ADMIN}, "rate_limit": 100}
    }
    
    def check_permission(
        self,
        tool_name: str,
        user_role: UserRole,
        user_id: str = None
    ) -> bool:
        """检查用户是否有权限调用工具"""
        if tool_name not in self.TOOL_PERMISSIONS:
            return False
        
        config = self.TOOL_PERMISSIONS[tool_name]
        
        # 角色检查
        if user_role not in config["roles"]:
            return False
        
        # 细粒度权限检查(可选)
        if tool_name == "delete_file" and user_id:
            # 检查是否是文件所有者
            # return self._is_file_owner(user_id)
            pass
        
        return True
    
    def check_rate_limit(self, tool_name: str, user_id: str) -> bool:
        """检查调用频率限制"""
        if tool_name not in self.TOOL_PERMISSIONS:
            return False
        
        rate_limit = self.TOOL_PERMISSIONS[tool_name].get("rate_limit")
        if not rate_limit:
            return True
        
        # 检查Redis中的计数器
        # current_count = redis.get(f"rate:{tool_name}:{user_id}")
        # return current_count < rate_limit
        return True


class SecureToolRegistry:
    """安全的工具注册表"""
    
    def __init__(self):
        self.tools = {}
        self.permission_manager = ToolPermissionManager()
    
    def execute(
        self,
        tool_name: str,
        user_role: UserRole,
        user_id: str,
        **params
    ) -> dict:
        """安全执行工具"""
        # 1. 权限检查
        if not self.permission_manager.check_permission(tool_name, user_role, user_id):
            return {
                "success": False,
                "error": "权限不足",
                "error_code": "PERMISSION_DENIED"
            }
        
        # 2. 频率限制检查
        if not self.permission_manager.check_rate_limit(tool_name, user_id):
            return {
                "success": False,
                "error": "调用频率超限",
                "error_code": "RATE_LIMIT_EXCEEDED"
            }
        
        # 3. 参数安全检查
        if not self._validate_params(tool_name, params):
            return {
                "success": False,
                "error": "参数不合法",
                "error_code": "INVALID_PARAMS"
            }
        
        # 4. 执行工具
        try:
            tool = self.tools[tool_name]
            result = tool(**params)
            
            return {
                "success": True,
                "result": result
            }
        
        except Exception as e:
            return {
                "success": False,
                "error": str(e),
                "error_code": "EXECUTION_ERROR"
            }
    
    def _validate_params(self, tool_name: str, params: dict) -> bool:
        """参数安全验证"""
        # 防止路径遍历攻击
        if "file_path" in params:
            if ".." in params["file_path"] or params["file_path"].startswith("/"):
                return False
        
        # 防止SQL注入
        if "query" in params:
            dangerous_patterns = ["DROP", "DELETE", "TRUNCATE", "--", ";"]
            if any(pattern in params["query"].upper() for pattern in dangerous_patterns):
                return False
        
        return True


# 使用示例
registry = SecureToolRegistry()
registry.register("delete_file", delete_file_tool)

### 4.6.3 审计日志

```python
from datetime import datetime
import json

class ToolAuditLogger:
    """工具调用审计日志"""
    
    def __init__(self, log_file: str = "./audit.log"):
        self.log_file = log_file
    
    def log(
        self,
        tool_name: str,
        user_id: str,
        user_role: str,
        params: dict,
        result: dict,
        execution_time: float
    ):
        """记录审计日志"""
        entry = {
            "timestamp": datetime.now().isoformat(),
            "tool_name": tool_name,
            "user_id": user_id,
            "user_role": user_role,
            "params": self._sanitize_params(params),
            "success": result.get("success", False),
            "error_code": result.get("error_code"),
            "execution_time_ms": execution_time * 1000
        }
        
        with open(self.log_file, 'a', encoding='utf-8') as f:
            f.write(json.dumps(entry, ensure_ascii=False) + '\n')
    
    def _sanitize_params(self, params: dict) -> dict:
        """脱敏敏感参数"""
        sensitive_keys = ["password", "token", "secret", "api_key"]
        
        sanitized = {}
        for key, value in params.items():
            if any(sk in key.lower() for sk in sensitive_keys):
                sanitized[key] = "***REDACTED***"
            else:
                sanitized[key] = value
        
        return sanitized
